% Sınıf etiketleri

% MDWT için seviye sayısı (7 seviyeli DWT)

% Özellik matrisini ve etiketleri saklamak için hücre dizileri

all_labels = [];

% Her klasördeki ses dosyalarını işleme
for class_idx = 1:num_classes
    folder = folders{class_idx};
    if ~exist(folder, 'dir')
        fprintf('HATA: Klasör bulunamadı: %s\n', folder);
        continue;
    end
    
    files = dir(fullfile(folder, '*.flac'));
    if isempty(files)
        fprintf('UYARI: Klasörde .wav dosyası yok: %s\n', folder);
        continue;
    end
    
    fprintf('İşleniyor: %s (%d dosya bulundu)\n', class_labels{class_idx}, length(files));
    
    for file_idx = 1:length(files)
        file_path = fullfile(folder, files(file_idx).name);
        fprintf('Dosya: %s\n', file_path);
        
        try
            % Ses dosyasını oku
            fid = fopen(file_path, 'r');
            if fid == -1
                error('Dosya açılamadı!');
            end
            fseek(fid, 44, -1); % WAV başlığını atla
            audio = fread(fid, inf, 'int16');
            fclose(fid);
            fs = 44100; % Varsayılan Fs
            audio = double(audio) / max(abs(double(audio)));
            
            % Sinyal uzunluğunu kontrol et (9 seviye için minimum 512 örnek)
            if length(audio) < 2^num_levels
                error('Sinyal çok kısa (%d < %d)!', length(audio), 2^num_levels);
            end
            
            fprintf('  - Okundu: Uzunluk=%d, Fs=%d Hz\n', length(audio), fs);
            
            % 9 Seviyeli MDWT Uygulama
            wavelet_name = 'sym4';
            [C, L] = wavedec(audio, num_levels, wavelet_name);
            
            % 9 Düşük Geçişli Filtre Bandını (Approximation Katsayıları) Elde Etme (D1-D9)
            bands = cell(num_levels, 1);
            for level = 1:num_levels
                approx_coeffs = appcoef(C, L, wavelet_name, level);
                bands{level} = approx_coeffs;
                fprintf('  - Seviye %d: Düşük geçişli bant uzunluğu=%d\n', level, length(approx_coeffs));
            end
            
            % Her bant için TSK-LHP ile özellik çıkarma
            band_features = [];
            for level = 1:num_levels
                signal = bands{level};
                features = TSK_LHP(signal); % TSK-LHP ile özellik çıkar
                fprintf('  - Seviye %d: Özellik vektörü oluşturuldu, uzunluk: %d\n', level, length(features));
                band_features = [band_features, features];
            end

            all_features = [all_features; band_features];
            all_labels = [all_labels; class_idx];

            fprintf('  - HATA: %s\n', ME.message);
            continue;
        end
    end

    fprintf('Chi2 Uygulama %d...\n', chi2_iter);

    for level = 1:num_levels
        hist_data = level_hist_features{level};

        chi2_stats = zeros(1, num_features);
        expected = mean(hist_data, 1);
        for f = 1:num_features
            observed = hist_data(:, f);
            if expected(f) == 0
                chi2_stats(f) = 0;
            else
                chi2_stats(f) = sum(((observed - expected(f)).^2) ./ expected(f));
            end

        [~, idx_chi2] = sort(chi2_stats, 'descend');
        selected_idx_chi2 = idx_chi2(1:num_features_per_level);

% Nested function: TSK-LHP feature extractor
function features = TSK_LHP(signal)
    window_size = 16;
    num_bins = 256;
    
    signal_length = length(signal);
    if signal_length < window_size
        fprintf('    UYARI: Sinyal çok kısa (%d < %d)\n', signal_length, window_size);
        features = zeros(1, num_bins * 3); % Signum, upper ternary, lower ternary
        return;
    end
    
    % Adım 2.1: Sinyali 16'lık örtüşen pencerelere böl
    num_windows = signal_length - window_size + 1;
    signum_values = zeros(num_windows, 1);
    upper_ternary_values = zeros(num_windows, 1);
    lower_ternary_values = zeros(num_windows, 1);
    
    % Adım 2.2: Eşik değerini hesapla (standart sapmanın yarısı)
    thresh = std(signal) / 2;
    
    for i = 1:num_windows
        window = signal(i:i+window_size-1);
        
        % Signum ve ternary bitlerini hesapla
        signum_bits = zeros(8, 1);
        upper_ternary_bits = zeros(8, 1);
        lower_ternary_bits = zeros(8, 1);
        
        for j = 1:8
            o = window(j);
            d = window(j+8);
            
            % Signum fonksiyonu
            signum_bits(j) = signum(o, d);
            
            % Ternary fonksiyonu
            tv = ternary(o, d, thresh);
            upper_ternary_bits(j) = upper_ternary_bit(tv);
            lower_ternary_bits(j) = lower_ternary_bit(tv);
        end
        
        % Adım 2.3: Bitlerden ondalık değerlere dönüştür
        signum_values(i) = sum(signum_bits .* (2.^(0:7)'));
        upper_ternary_values(i) = sum(upper_ternary_bits .* (2.^(0:7)'));
        lower_ternary_values(i) = sum(lower_ternary_bits .* (2.^(0:7)'));
    end
    
    % Adım 2.4: Histogramları çıkar
    hist_signum = histcounts(signum_values, 0:num_bins, 'Normalization', 'probability');
    hist_upper = histcounts(upper_ternary_values, 0:num_bins, 'Normalization', 'probability');
    hist_lower = histcounts(lower_ternary_values, 0:num_bins, 'Normalization', 'probability');
    
    % Adım 2.5: Histogramları birleştir
    features = [hist_signum, hist_upper, hist_lower]; % 768 uzunluğunda (256 x 3)
end

% Signum fonksiyonu
function bit = signum(o, d)
    if (o - d) < 0
        bit = 0;
    else
        bit = 1;
    end
end

% Ternary fonksiyonu
function tv = ternary(o, d, thresh)
    diff = o - d;
    if diff < -thresh
        tv = -1;
    elseif diff > thresh
        tv = 1;
    else
        tv = 0;
    end
end

% Upper ternary bit
function bit = upper_ternary_bit(tv)
    if tv == 1
        bit = 1;
    else
        bit = 0;
    end
end

% Lower ternary bit
function bit = lower_ternary_bit(tv)
    if tv == -1
        bit = 1;
    else
        bit = 0;
    end
end
